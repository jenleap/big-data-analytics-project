import pandas as pd
import numpy as np
import networkx as nx
from community import community_louvain  # pip install python-louvain

def build_network_and_validate(substitutes_df, pairwise_df, top_n=5,
                               lift_threshold=3, support_threshold=5e-7):
    """
    Build a complement network from pairwise probabilities, detect clusters,
    and validate hybrid score vs substitution_index using network-based metrics.

    Parameters
    ----------
    substitutes_df : pd.DataFrame
        DataFrame with columns ['product_id', 'substitute_id', 'score', 'substitution_index']
    pairwise_df : pd.DataFrame
        DataFrame with columns ['product_i', 'product_j', 'P_i', 'P_j', 'P_ij']
    top_n : int
        Number of top substitutes to evaluate per product
    lift_threshold : float
        Minimum lift to keep edge
    support_threshold : float
        Minimum P_ij to keep edge

    Returns
    -------
    pivot_df : pd.DataFrame
        Per-product comparison of metrics
    aggregate_summary : pd.DataFrame
        Summary count of which metric is better
    G : networkx.Graph
        The constructed complement network with clusters
    """

    # 1️⃣ Compute lift
    pairwise_df = pairwise_df.copy()
    pairwise_df['lift'] = pairwise_df.apply(
        lambda x: x['P_ij'] / (x['P_i'] * x['P_j']) if x['P_i'] * x['P_j'] > 0 else 0,
        axis=1
    )

    # 2️⃣ Filter edges by lift and support
    filtered = pairwise_df[
        (pairwise_df['lift'] > lift_threshold) &
        (pairwise_df['P_ij'] > support_threshold)
    ].copy()

    # 3️⃣ Build network
    G = nx.Graph()
    for _, row in filtered.iterrows():
        G.add_edge(row['product_i'], row['product_j'], weight=row['lift'])

    # 4️⃣ Detect Louvain clusters
    partition = community_louvain.best_partition(G, weight='weight')
    nx.set_node_attributes(G, partition, 'cluster')

    # 5️⃣ Precompute neighbors
    neighbors_dict = {n: set(G.neighbors(n)) for n in G.nodes}
    cluster_dict = partition

    results = []

    for product in substitutes_df['product_id'].unique():
        product_subs = substitutes_df[substitutes_df['product_id'] == product]

        if product not in neighbors_dict:
            continue

        focus_neighbors = neighbors_dict[product]

        for metric_col in ['score', 'substitution_index']:
            top_subs = product_subs.sort_values(metric_col, ascending=False).head(top_n)['substitute_id']

            neighbor_sims = []
            same_cluster_flags = []
            path_lengths = []

            for sub in top_subs:
                # Skip if substitute not in network
                if sub not in G:
                    neighbor_sims.append(0)
                    same_cluster_flags.append(0)
                    path_lengths.append(0)
                    continue

                sub_neighbors = neighbors_dict.get(sub, set())

                # Neighborhood similarity: Jaccard of neighbors
                union = focus_neighbors | sub_neighbors
                intersection = focus_neighbors & sub_neighbors
                neighbor_sims.append(len(intersection) / len(union) if union else 0)

                # Cluster alignment
                same_cluster = int(cluster_dict.get(product) == cluster_dict.get(sub))
                same_cluster_flags.append(same_cluster)

                # Shortest path length (inverse)
                try:
                    path_len = nx.shortest_path_length(G, source=product, target=sub)
                    path_lengths.append(1 / path_len if path_len > 0 else 1.0)
                except nx.NetworkXNoPath:
                    path_lengths.append(0)

            results.append({
                'product_id': product,
                'metric': metric_col,
                'avg_neighbor_similarity': np.mean(neighbor_sims),
                'cluster_alignment': np.mean(same_cluster_flags),
                'avg_inverse_path': np.mean(path_lengths)
            })

    validation_df = pd.DataFrame(results)

    # Pivot for comparison
    pivot_df = validation_df.pivot(index='product_id', columns='metric').reset_index()
    pivot_df.columns = ['product_id',
                        'avg_neighbor_similarity_hybrid', 'avg_neighbor_similarity_index',
                        'cluster_alignment_hybrid', 'cluster_alignment_index',
                        'avg_inverse_path_hybrid', 'avg_inverse_path_index']

    # Decide better metric per product
    def choose_better(row):
        score = ((row['avg_neighbor_similarity_hybrid'] - row['avg_neighbor_similarity_index']) +
                 (row['cluster_alignment_hybrid'] - row['cluster_alignment_index']) +
                 (row['avg_inverse_path_hybrid'] - row['avg_inverse_path_index']))
        return 'hybrid' if score > 0 else 'substitution_index'

    pivot_df['better_metric'] = pivot_df.apply(choose_better, axis=1)

    # Aggregate summary
    aggregate_summary = pivot_df['better_metric'].value_counts().reset_index()
    aggregate_summary.columns = ['metric', 'num_products_better']

    return pivot_df, aggregate_summary, G
